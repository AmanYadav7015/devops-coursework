```console
root@ubuntu-hw:~# cd /root && cp sysinfo.sh sysinfo_noexec.sh && chmod -x sysinfo_noexec.sh && ls -l sysinfo_noexec.sh
-rw-r--r-- 1 root root 4911 Sep  2 17:26 sysinfo_noexec.sh

# --- attempt to run it without the execute bit set (real failure) ---
root@ubuntu-hw:~# cd /root && ./sysinfo_noexec.sh
bash: line 1: ./sysinfo_noexec.sh: Permission denied

# --- fix: grant execute permission ---
root@ubuntu-hw:~# cd /root && chmod +x sysinfo_noexec.sh && ls -l sysinfo_noexec.sh
-rwxr-xr-x 1 root root 4911 Sep  2 17:26 sysinfo_noexec.sh

# --- confirm the fix works (quick piped run) ---
root@ubuntu-hw:~# cd /root && echo "permfix_test" | ./sysinfo_noexec.sh > /dev/null && echo "Ran successfully after chmod +x" && rm -rf sysinfo_noexec.sh permfix_test
Ran successfully after chmod +x
```
